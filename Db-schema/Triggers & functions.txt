CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN

    NEW.updated_at := CURRENT_TIMESTAMP;

    RETURN NEW;

END;
$$;

CREATE OR REPLACE FUNCTION fn_audit_logger()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
DECLARE

    v_module_name VARCHAR;

    v_record_id BIGINT;

    v_pk_column VARCHAR;

BEGIN

    ------------------------------------------------------------------
    -- Module Name (Passed from Trigger)
    ------------------------------------------------------------------

    v_module_name := TG_ARGV[0];

    ------------------------------------------------------------------
    -- Build Primary Key Column Name
    ------------------------------------------------------------------

    v_pk_column :=
        CASE TG_TABLE_NAME

            WHEN 'users' THEN 'user_id'

            WHEN 'goals' THEN 'goal_id'

            WHEN 'goal_progress' THEN 'progress_id'

            WHEN 'incomes' THEN 'income_id'

            WHEN 'expenses' THEN 'expense_id'

            WHEN 'budgets' THEN 'budget_id'

            WHEN 'user_subscriptions' THEN 'subscription_id'

            WHEN 'customer_reviews' THEN 'review_id'

            WHEN 'otp_verifications' THEN 'otp_id'

            WHEN 'social_logins' THEN 'social_login_id'

            WHEN 'user_sessions' THEN 'session_id'

            WHEN 'stock_data' THEN 'stock_data_id'

            WHEN 'technical_indicators' THEN 'technical_indicator_id'

            WHEN 'company_news' THEN 'news_id'

            WHEN 'company_sentiments' THEN 'sentiment_id'

            ELSE NULL

        END;

    ------------------------------------------------------------------
    -- INSERT
    ------------------------------------------------------------------

    IF TG_OP = 'INSERT' THEN

        v_record_id := (to_jsonb(NEW)->>v_pk_column)::BIGINT;

        INSERT INTO audit_logs
        (
            user_id,
            module_name,
            table_name,
            record_id,
            action,
            old_data,
            new_data,
            description
        )
        VALUES
        (
            NEW.user_id,
            v_module_name,
            TG_TABLE_NAME,
            v_record_id,
            TG_OP,
            NULL,
            to_jsonb(NEW),
            TG_OP || ' on ' || TG_TABLE_NAME
        );

        RETURN NEW;

    END IF;

    ------------------------------------------------------------------
    -- UPDATE
    ------------------------------------------------------------------

    IF TG_OP = 'UPDATE' THEN

        v_record_id := (to_jsonb(NEW)->>v_pk_column)::BIGINT;

        INSERT INTO audit_logs
        (
            user_id,
            module_name,
            table_name,
            record_id,
            action,
            old_data,
            new_data,
            description
        )
        VALUES
        (
            NEW.user_id,
            v_module_name,
            TG_TABLE_NAME,
            v_record_id,
            TG_OP,
            to_jsonb(OLD),
            to_jsonb(NEW),
            TG_OP || ' on ' || TG_TABLE_NAME
        );

        RETURN NEW;

    END IF;

    ------------------------------------------------------------------
    -- DELETE
    ------------------------------------------------------------------

    IF TG_OP = 'DELETE' THEN

        v_record_id := (to_jsonb(OLD)->>v_pk_column)::BIGINT;

        INSERT INTO audit_logs
        (
            user_id,
            module_name,
            table_name,
            record_id,
            action,
            old_data,
            new_data,
            description
        )
        VALUES
        (
            OLD.user_id,
            v_module_name,
            TG_TABLE_NAME,
            v_record_id,
            TG_OP,
            to_jsonb(OLD),
            NULL,
            TG_OP || ' on ' || TG_TABLE_NAME
        );

        RETURN OLD;

    END IF;

    RETURN NULL;

END;
$$;

CREATE TRIGGER trg_users_audit
AFTER INSERT OR UPDATE OR DELETE
ON users
FOR EACH ROW
EXECUTE FUNCTION fn_audit_logger(
    'Authentication',
    'user_id'
);
